import numpy as np
import matplotlib.pyplot as plt

def simulate_handoff(distances, rssi_1, rssi_2, hysteresis_db, ping_pong_window_m=50.0):
    serving_bs = 1
    handoff_distances = []
    ping_pong_count = 0
    
    for d, s1, s2 in zip(distances, rssi_1, rssi_2):
        if serving_bs == 1 and (s2 > s1 + hysteresis_db):
            if handoff_distances and (d - handoff_distances[-1] <= ping_pong_window_m):
                ping_pong_count += 1
            serving_bs = 2
            handoff_distances.append(d)
        elif serving_bs == 2 and (s1 > s2 + hysteresis_db):
            if handoff_distances and (d - handoff_distances[-1] <= ping_pong_window_m):
                ping_pong_count += 1
            serving_bs = 1
            handoff_distances.append(d)
            
    return len(handoff_distances), ping_pong_count, handoff_distances

if __name__ == "__main__":
    np.random.seed(42)
    
    D = 1000.0
    d0 = 10.0
    Pt = 30.0
    L0 = 40.0
    n = 3.5
    sigma = 4.0
    
    distances = np.linspace(20, D - 20, 200)
    
    shadow_1 = np.random.normal(0, sigma, size=len(distances))
    shadow_2 = np.random.normal(0, sigma, size=len(distances))
    
    rssi_1 = Pt - L0 - 10 * n * np.log10(distances / d0) + shadow_1
    rssi_2 = Pt - L0 - 10 * n * np.log10((D - distances) / d0) + shadow_2
    
    h_margins = [0, 2, 4, 6, 8, 10, 12]
    total_handoffs = []
    ping_pongs = []
    
    print(f"Simulation Parameters: BS Separation = {D} m, Path Loss Exp (n) = {n}, Shadowing (sigma) = {sigma} dB\n")
    print(f"{'Hysteresis H (dB)':<18} | {'Total Handoffs':<16} | {'Ping-Pong Events':<18} | {'Handoff Locations (m)'}")
    print("-" * 82)
    
    ho_locs_h6 = []
    for H in h_margins:
        ho_count, pp_count, ho_locs = simulate_handoff(distances, rssi_1, rssi_2, H)
        total_handoffs.append(ho_count)
        ping_pongs.append(pp_count)
        if H == 6:
            ho_locs_h6 = ho_locs
        locs_str = ", ".join([f"{loc:.0f}" for loc in ho_locs[:4]]) + ("..." if len(ho_locs) > 4 else "")
        print(f"{H:<18d} | {ho_count:<16d} | {pp_count:<18d} | {locs_str}")
        
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))
    
    ax1.plot(distances, rssi_1, label="RSSI BS1", color="tab:blue", alpha=0.85)
    ax1.plot(distances, rssi_2, label="RSSI BS2", color="tab:orange", alpha=0.85)
    for idx, loc in enumerate(ho_locs_h6):
        ax1.axvline(x=loc, color="green", linestyle="--", linewidth=1.5, label="Handoff (H = 6 dB)" if idx == 0 else "")
    ax1.set_title("RSSI Profiles & Handoff Events (H = 6 dB)")
    ax1.set_xlabel("Distance from BS1 (m)")
    ax1.set_ylabel("Received Signal Strength (dBm)")
    ax1.grid(True, linestyle="--", alpha=0.6)
    ax1.legend()
    
    ax2.plot(h_margins, total_handoffs, marker="o", color="tab:red", linewidth=2, label="Total Handoffs")
    ax2.plot(h_margins, ping_pongs, marker="s", linestyle="--", color="purple", linewidth=2, label="Ping-Pong Handoffs")
    ax2.set_title("Total Handoffs & Ping-Pong Events vs. Hysteresis Margin")
    ax2.set_xlabel("Hysteresis Margin H (dB)")
    ax2.set_ylabel("Number of Events")
    ax2.grid(True, linestyle="--", alpha=0.6)
    ax2.legend()
    
    plt.tight_layout()
    plt.show()
